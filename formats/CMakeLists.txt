cmake_minimum_required(VERSION 3.21)
project(SoniclawFormats VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CLAP_WRAPPER_DOWNLOAD_DEPENDENCIES OFF CACHE BOOL "Use pinned local SDKs")
set(CLAP_WRAPPER_COPY_AFTER_BUILD OFF CACHE BOOL "Never install as a build side effect")
set(CLAP_WRAPPER_BUILD_AAX OFF CACHE BOOL "No AAX integration")
if(NOT DEFINED SONICLAW_DEPS OR NOT DEFINED SONICLAW_STATIC_LIBRARY)
  message(FATAL_ERROR "Use tools/build-formats.py, or supply SONICLAW_DEPS and SONICLAW_STATIC_LIBRARY")
endif()
set(CLAP_SDK_ROOT "${SONICLAW_DEPS}/clap" CACHE PATH "CLAP headers")
set(VST3_SDK_ROOT "${SONICLAW_DEPS}/vst3sdk" CACHE PATH "VST3 SDK")
set(AUDIOUNIT_SDK_ROOT "${SONICLAW_DEPS}/AudioUnitSDK" CACHE PATH "AUv2 SDK")
add_subdirectory("${SONICLAW_DEPS}/clap-wrapper" wrapper EXCLUDE_FROM_ALL)
# Read the Rust CLAP entry directly. No separate installed .clap is needed.
target_compile_definitions(clap-wrapper-shared-detail PRIVATE STATICALLY_LINKED_CLAP_ENTRY=1)
add_library(soniclaw_impl STATIC IMPORTED GLOBAL)
set_target_properties(soniclaw_impl PROPERTIES IMPORTED_LOCATION "${SONICLAW_STATIC_LIBRARY}")
add_library(soniclaw_native INTERFACE)
if(APPLE)
  target_link_libraries(soniclaw_native INTERFACE "-framework AppKit" "-framework CoreFoundation" "-framework CoreGraphics" "-framework Foundation" "-framework OpenGL" "-framework Security" "-framework SystemConfiguration" "-framework CoreVideo" "-framework IOKit" c++ iconv)
elseif(WIN32)
  target_link_libraries(soniclaw_native INTERFACE user32 gdi32 opengl32 shell32 ole32 oleaut32 advapi32 bcrypt userenv ntdll ws2_32 dbghelp comctl32 dwmapi imm32 version winmm)
else()
  target_link_libraries(soniclaw_native INTERFACE dl pthread m rt util X11 X11-xcb xcb GL)
endif()
add_library(soniclaw_vst3 MODULE entry.cpp)
target_add_vst3_wrapper(TARGET soniclaw_vst3 OUTPUT_NAME "SoniclawReference" BUNDLE_IDENTIFIER "org.sonora.reference.vst3" BUNDLE_VERSION "0.3.0")
target_link_libraries(soniclaw_vst3 PRIVATE soniclaw_impl soniclaw_native)
if(APPLE AND SONICLAW_BUILD_AU)
  enable_language(OBJC)
  enable_language(OBJCXX)
  # An AU component has one product identity. This wrapper chooses the first
  # CLAP descriptor (Gain + Filter); CLAP/VST3 expose the whole factory.
  add_library(soniclaw_auv2 MODULE entry.cpp)
  target_add_auv2_wrapper(TARGET soniclaw_auv2 OUTPUT_NAME "SoniclawReference" BUNDLE_IDENTIFIER "org.sonora.reference.au" BUNDLE_VERSION "0.3.0" MANUFACTURER_NAME "Soniclaw" MANUFACTURER_CODE "Snra" SUBTYPE_CODE "SnGF" INSTRUMENT_TYPE "aufx" PREFER_CMAKE_AUV2_CONFIGURATION TRUE)
  target_link_libraries(soniclaw_auv2 PRIVATE soniclaw_impl soniclaw_native)
endif()
